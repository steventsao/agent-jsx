// One provider owns the goal. Children only move the machine
// through the explicit grant: result(handleGoalEvent).
export const ShipRelease = composeAgent(
  <GoalProvider name="ship-v0.6" spec={releaseSpec}>
    {({ phase, plan, criteria, incident, handleGoalEvent }) => (
      <>
        <Sensor name="uptime:docs" url={DOCS_URL} every="30s"
                onDown={result(handleGoalEvent)} />
        {phase === "assess" && // deterministic: is the spec verifiable?
          <AssessSpec spec={releaseSpec}
                      onResult={result(handleGoalEvent)} />}
        {phase === "plan" && // model agent decomposes the goal
          <Agent agentClass={PlannerAgent} spec={releaseSpec}
                 onPlan={result(handleGoalEvent)} />}
        {phase === "execute" && plan.milestones.map((m) =>
          m.kind === "agent"
            ? <Agent key={m.id} agentClass={WorkerAgent} task={m}
                     onResult={result(handleGoalEvent)} />
            : <Step key={m.id} run={m.run} // deterministic code
                    onResult={result(handleGoalEvent)} />)}
        {phase === "verify" && criteria.map((c) =>
          <Verifier key={c.id} criterion={c} // det | sensor | agent
                    onCheck={result(handleGoalEvent)} />)}
        {phase === "review" && // human-in-review gate
          <HumanGate name="review:owner" summary={summarize(criteria)}
                     onDecision={result(handleGoalEvent)} />}
        {phase === "repair" &&
          <Agent agentClass={RepairAgent} incident={incident}
                 onResult={result(handleGoalEvent)} />}
      </>
    )}
  </GoalProvider>,
);

// goal-provider.tsx — the deterministic core
export class GoalProvider extends Agent<GoalState, GoalProps> {
  static agentName = "goal-provider";
  initialState = idleGoalState;

  render() {
    return this.define({
      model: "openrouter/openai/gpt-5-mini",
      description: "Owns one long-horizon goal as a state machine.",
    });
  }

  @callable()
  handleGoalEvent(event: GoalEvent): void {
    // every edge in the diagram is one branch of this pure reducer
    this.setState((state) => reduceGoal(state, event));
  }
}
